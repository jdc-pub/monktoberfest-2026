function GPT3(prompt::String; filename, kwargs...)
    response = let
      choice = first( <1>
        openai.Completion.create(; kwargs..., n=1)["choices"])
      response = choice["text"] |> string
    end

    open(filename, "w") do io
        write(io,
            """
            This file was autogenerated with GPT-3! All content <2>

            produce any inappropriate content. Still, if you
            find ANY offensive content in this file, please
            contact the project's maintainer (@cadojo on GitHub).

            This file has the following format: <prompt>... <response>

            ---

            $prompt... $response <3>

            """
        )
    end

    return response
end

== 2023: Plagiarism via ChatGPT

* *Fall 2022 Discussion Post*
** _I agree with Mark's comment about the text lacking pathos._

[.fragment]
* *Spring 2023 Discussion Post*
** _I see what you mean, but I'd push back a little on the idea that the text lacks pathos — I think the emotional appeal is there, just operating more quietly than we might expect. Pathos doesn't have to mean vivid sob stories or dramatic language; sometimes an author builds emotion through understatement, word choice, or even a deliberately cold, clinical tone that makes the reader feel *more* unsettled, not less. When you say the piece feels emotionally flat, I'd encourage you to ask whether that flatness is an absence of pathos or a strategic choice aimed at a specific audience—maybe readers who'd distrust anything that feels too manipulative. For me, the most affecting moment was actually the most restrained one, which suggests the author trusts the audience to supply the feeling themselves._

[.notes]
--
